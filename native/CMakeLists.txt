cmake_minimum_required(VERSION 3.16)
project(FancyTasksNGNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Test)
add_library(fancytasksnative MODULE WindowRegion.cpp IconIdentity.h)
target_link_libraries(fancytasksnative PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
install(TARGETS fancytasksnative LIBRARY DESTINATION lib/fancytasksng/native)
install(FILES qmldir plugins.qmltypes DESTINATION lib/fancytasksng/native)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_icon_identity tests/test_icon_identity.cpp IconIdentity.h)
    target_include_directories(test_icon_identity PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(test_icon_identity PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
    add_test(NAME icon-identity COMMAND test_icon_identity)
    set_tests_properties(icon-identity PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(test_window_region test_window_region.cpp)
    target_link_libraries(test_window_region PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
    add_test(NAME window-region COMMAND test_window_region)
    set_tests_properties(window-region PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(test_zoom_hit_target test_zoom_hit_target.cpp)
    target_link_libraries(test_zoom_hit_target PRIVATE Qt6::Quick Qt6::Qml Qt6::Test)
    target_compile_definitions(test_zoom_hit_target PRIVATE HIT_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/test_hit_target.qml")
    target_compile_definitions(test_zoom_hit_target PRIVATE REGION_PLUGIN="$<TARGET_FILE:fancytasksnative>")
    add_dependencies(test_zoom_hit_target fancytasksnative)
    add_test(NAME zoom-hit-target COMMAND test_zoom_hit_target)
    set_tests_properties(zoom-hit-target PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
